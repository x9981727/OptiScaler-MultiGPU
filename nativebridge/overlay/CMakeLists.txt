cmake_minimum_required(VERSION 3.20)
project(MagpieNativeBridge VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NB_SANITIZE "Enable address/undefined sanitizers on GCC/Clang" OFF)
add_library(native_bridge_core src/contract.cpp src/ledger.cpp src/wire.cpp)
target_include_directories(native_bridge_core PUBLIC include)
if(MSVC)
  target_compile_options(native_bridge_core PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(native_bridge_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(NB_SANITIZE)
    target_compile_options(native_bridge_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(native_bridge_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
find_package(Threads REQUIRED)
target_link_libraries(native_bridge_core PUBLIC Threads::Threads)
add_executable(native_bridge_tests tests/core_tests.cpp)
target_link_libraries(native_bridge_tests PRIVATE native_bridge_core)
add_executable(native_bridge_bench tests/benchmark.cpp)
target_link_libraries(native_bridge_bench PRIVATE native_bridge_core)
enable_testing()
add_test(NAME native_bridge_core COMMAND native_bridge_tests)
if(WIN32)
  add_library(native_bridge_d3d12 STATIC
    windows/transport_d3d12.cpp
    windows/ipc_pipe.cpp
    windows/local_interop.cpp
    windows/session_protocol.cpp)
  target_include_directories(native_bridge_d3d12 PUBLIC windows)
  target_link_libraries(native_bridge_d3d12 PUBLIC native_bridge_core d3d12 d3d11 dxgi advapi32)
  target_compile_definitions(native_bridge_d3d12 PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  if(MSVC)
    target_compile_options(native_bridge_d3d12 PRIVATE /W4 /WX /permissive-)
  endif()
  add_executable(native_bridge_ipc_selftest windows/ipc_selftest.cpp)
  target_include_directories(native_bridge_ipc_selftest PRIVATE tests)
  target_link_libraries(native_bridge_ipc_selftest PRIVATE native_bridge_d3d12)
  add_test(NAME windows_two_process_ipc COMMAND native_bridge_ipc_selftest)
  set_tests_properties(windows_two_process_ipc PROPERTIES TIMEOUT 30)
  add_executable(native_bridge_session_protocol_tests windows/session_protocol_tests.cpp)
  target_include_directories(native_bridge_session_protocol_tests PRIVATE tests)
  target_link_libraries(native_bridge_session_protocol_tests PRIVATE native_bridge_d3d12)
  add_test(NAME windows_session_protocol COMMAND native_bridge_session_protocol_tests)
  add_executable(native_bridge_adapters windows/adapters.cpp)
  target_link_libraries(native_bridge_adapters PRIVATE dxgi)
  add_executable(native_bridge_d3d_runtime_selftest windows/d3d_runtime_selftest.cpp)
  target_link_libraries(native_bridge_d3d_runtime_selftest PRIVATE native_bridge_d3d12 d3d11 d3d12 dxgi)
  add_test(NAME windows_d3d_runtime_interop COMMAND native_bridge_d3d_runtime_selftest)
  set_tests_properties(windows_d3d_runtime_interop PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
endif()
add_executable(magpie_guidance_tests tests/magpie_guidance_tests.cpp)
target_include_directories(magpie_guidance_tests PRIVATE patches tests/mocks)
target_link_libraries(magpie_guidance_tests PRIVATE native_bridge_core)
add_test(NAME patched_magpie_guidance COMMAND magpie_guidance_tests)
add_executable(streamline_camera_tests tests/streamline_camera_tests.cpp)
target_link_libraries(streamline_camera_tests PRIVATE native_bridge_core)
add_test(NAME streamline_camera_adapter COMMAND streamline_camera_tests)
add_executable(native_bridge_wire_tests tests/wire_tests.cpp)
target_link_libraries(native_bridge_wire_tests PRIVATE native_bridge_core)
add_test(NAME wire_packet COMMAND native_bridge_wire_tests)
