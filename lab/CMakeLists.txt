cmake_minimum_required(VERSION 3.20)
project(XeFGVirtualLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PRODUCTION_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../upstream/OptiScaler" CACHE PATH "Actual reconstructed OptiScaler sources")
if(NOT EXISTS "${PRODUCTION_ROOT}/framegen/ManagedXeFGSequence.h")
 message(FATAL_ERROR "Reconstruct v25 and apply sequence-only.patch first")
endif()
include(CTest)
add_executable(protocol-stress protocol_stress.cpp)
target_include_directories(protocol-stress PRIVATE "${PRODUCTION_ROOT}/framegen")
add_executable(sequence-regression ../v25-sequence/sequence-regression.cpp)
target_include_directories(sequence-regression PRIVATE "${PRODUCTION_ROOT}/framegen")
add_test(NAME protocol-stress COMMAND protocol-stress)
add_test(NAME sequence-regression COMMAND sequence-regression)
set_tests_properties(protocol-stress sequence-regression PROPERTIES TIMEOUT 120)
if(MSVC)
 foreach(t protocol-stress sequence-regression)
  target_compile_options(${t} PRIVATE /W4 /WX /EHsc)
 endforeach()
else()
 foreach(t protocol-stress sequence-regression)
  target_compile_options(${t} PRIVATE -Wall -Wextra -Werror)
 endforeach()
 option(SANITIZE "ASan and UBSan" ON)
 if(SANITIZE)
  foreach(t protocol-stress sequence-regression)
   target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endforeach()
 endif()
endif()
if(WIN32)
 add_executable(copy-stress copy_stress.cpp)
 target_include_directories(copy-stress PRIVATE "${PRODUCTION_ROOT}")
 target_compile_definitions(copy-stress PRIVATE NOMINMAX)
 target_link_libraries(copy-stress PRIVATE d3d12 dxgi)
 add_test(NAME full-resolution-transfer COMMAND copy-stress)
 set_tests_properties(full-resolution-transfer PROPERTIES TIMEOUT 240)
 add_executable(sdk-probe sdk_probe.cpp)
 target_include_directories(sdk-probe PRIVATE "${PRODUCTION_ROOT}/../external/xess/inc")
 target_compile_definitions(sdk-probe PRIVATE NOMINMAX)
 target_link_libraries(sdk-probe PRIVATE d3d12 dxgi user32)
 # A capability observation cannot replace a target-hardware acceptance test.
endif()
